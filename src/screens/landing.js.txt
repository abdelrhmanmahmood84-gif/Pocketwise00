
    (function initCalculator() {
      const incomeRange = document.getElementById('income-range');
      const incomeDisplay = document.getElementById('slider-income-display');
      const projectedAnnual = document.getElementById('projected-annual');
      const monthlySavedVal = document.getElementById('monthly-saved-val');
      const interestEarnedVal = document.getElementById('interest-earned-val');
      const runwayVal = document.getElementById('runway-val');
      const rateBtns = document.querySelectorAll('.rate-btn');

      let currentRate = 0.20;

      function updateCalc() {
        if (!incomeRange) return;
        const income = parseFloat(incomeRange.value);
        if (incomeDisplay) incomeDisplay.textContent = '$' + income.toLocaleString();

        const monthlySave = income * currentRate;
        const totalPrincipal = monthlySave * 12;
        const interest = totalPrincipal * (0.051 / 2);
        const annualTotal = totalPrincipal + interest;

        if (monthlySavedVal) monthlySavedVal.textContent = '$' + Math.round(monthlySave).toLocaleString() + '.00';
        if (interestEarnedVal) interestEarnedVal.textContent = '+$' + Math.round(interest).toLocaleString() + '.00';
        if (projectedAnnual) projectedAnnual.textContent = '$' + Math.round(annualTotal).toLocaleString();

        const runway = (annualTotal / (income * 0.7)).toFixed(1);
        if (runwayVal) runwayVal.textContent = runway + ' Months';
      }

      if (incomeRange) {
        incomeRange.addEventListener('input', updateCalc);
      }

      rateBtns.forEach(btn => {
        btn.addEventListener('click', () => {
          rateBtns.forEach(b => {
            b.classList.remove('bg-secondary-fixed', 'text-primary', 'font-bold');
            b.classList.add('bg-tertiary-container', 'text-tertiary-fixed');
          });
          btn.classList.add('bg-secondary-fixed', 'text-primary', 'font-bold');
          btn.classList.remove('bg-tertiary-container', 'text-tertiary-fixed');
          currentRate = parseFloat(btn.getAttribute('data-rate'));
          updateCalc();
        });
      });

      updateCalc();

      // Check if user is signed in to reflect live balance in the hero card
      (async function checkUserHeroBalance() {
        const sb = window.__pwSupabase;
        if (!sb) return;
        try {
          const { data: userData } = await sb.auth.getUser();
          if (!userData?.user) return;
          const { data: txs } = await sb
            .from('transactions')
            .select('transaction_type, type, amount')
            .eq('user_id', userData.user.id);

          if (txs && txs.length > 0) {
            let bal = 0;
            txs.forEach(t => {
              const isInc = (t.transaction_type || t.type) === 'income';
              const a = Number(t.amount) || 0;
              bal += isInc ? a : -a;
            });
            const el = document.getElementById('hero-balance-val');
            if (el) el.textContent = '$' + bal.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
          }
        } catch (e) {
          // Silent fallback to default showcase number
        }
      })();
    })();
